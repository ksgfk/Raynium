cmake_minimum_required(VERSION 3.25)
project(RayniumDebugDependency LANGUAGES C CXX)
include(GNUInstallDirs)
if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" OR
   NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64" OR
   NOT CMAKE_BUILD_TYPE STREQUAL "Debug" OR
   NOT CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "MultiThreadedDebug")
    message(FATAL_ERROR "Raynium Debug dependencies require Windows x64 MSVC /MTd.")
endif()
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0126 NEW)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
set(CMAKE_EXPORT_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
add_compile_options(/utf-8)
if(RAYNIUM_DEPENDENCY_NAME STREQUAL "stdexec")
    find_package(Threads REQUIRED)
    add_library(stdexec INTERFACE)
    set_target_properties(stdexec PROPERTIES EXPORT_NAME stdexec)
    target_compile_features(stdexec INTERFACE cxx_std_20)
    target_link_libraries(stdexec INTERFACE Threads::Threads)
    target_include_directories(stdexec INTERFACE "$<INSTALL_INTERFACE:include>")
    install(DIRECTORY "${RAYNIUM_DEPENDENCY_SOURCE}/include/" DESTINATION include)
    install(TARGETS stdexec EXPORT stdexec-targets)
    install(EXPORT stdexec-targets NAMESPACE STDEXEC:: DESTINATION lib/cmake/stdexec)
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/stdexecConfig.cmake" CONTENT
        "include(CMakeFindDependencyMacro)\nfind_dependency(Threads)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/stdexec-targets.cmake\")\n" @ONLY)
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/stdexecConfig.cmake" DESTINATION lib/cmake/stdexec)
else()
    if(RAYNIUM_DEPENDENCY_NAME STREQUAL "volk")
        find_package(VulkanHeaders CONFIG REQUIRED)
        get_target_property(includes Vulkan::Headers INTERFACE_INCLUDE_DIRECTORIES)
        include_directories(${includes})
    endif()
    add_subdirectory("${RAYNIUM_DEPENDENCY_SOURCE}" upstream)
endif()
